/* GAZELA SPINER SENSOR BLE TEST v5 Arduino Nano 33 BLE Sense Rev2 TEST: - BLE connection - LIVE DATA przez NOTIFY - COMMAND przez WRITE - komenda L uruchamia LIVE */ #include #include #include // ===================================================== // BLE UUID // ===================================================== // SENSOR DATA SERVICE const char* SENSOR_SERVICE_UUID = "7a1e0001-5a9a-4c71-9b3a-47415a5a0001"; // LIVE DATA CHARACTERISTIC const char* LIVE_CHARACTERISTIC_UUID = "7a1e0002-5a9a-4c71-9b3a-47415a5a0001"; // CONTROL SERVICE const char* CONTROL_SERVICE_UUID = "7a1e0010-5a9a-4c71-9b3a-47415a5a0001"; // COMMAND CHARACTERISTIC const char* COMMAND_CHARACTERISTIC_UUID = "7a1e0011-5a9a-4c71-9b3a-47415a5a0001"; // ===================================================== // BLE SERVICES // ===================================================== BLEService sensorService( SENSOR_SERVICE_UUID ); BLEService controlService( CONTROL_SERVICE_UUID ); // ===================================================== // BLE CHARACTERISTICS // ===================================================== // Arduino → telefon BLEStringCharacteristic liveCharacteristic( LIVE_CHARACTERISTIC_UUID, BLERead | BLENotify, 100 ); // Telefon → Arduino BLEStringCharacteristic commandCharacteristic( COMMAND_CHARACTERISTIC_UUID, BLEWrite, 20 ); // ===================================================== // USTAWIENIA // ===================================================== const unsigned long SAMPLE_INTERVAL = 100; unsigned long lastSampleTime = 0; // ===================================================== // STAN LIVE // ===================================================== bool liveMode = false; // ===================================================== // ANGLE // ===================================================== float angleY_Gyro = 0.0; unsigned long previousTime = 0; // ===================================================== // ODCZYT SENSORÓW // ===================================================== void readSensorData( float &AX, float &AY, float &AZ, float &G, float &Angle, float &GX, float &GY, float &GZ, float &AngleY ) { float sensorX; float sensorY; float sensorZ; float gyroX; float gyroY; float gyroZ; // --------------------------------------------------- // ACCELEROMETER // --------------------------------------------------- if (IMU.accelerationAvailable()) { IMU.readAcceleration( sensorX, sensorY, sensorZ ); } else { sensorX = 0; sensorY = 0; sensorZ = 0; } // --------------------------------------------------- // GYROSCOPE // --------------------------------------------------- if (IMU.gyroscopeAvailable()) { IMU.readGyroscope( gyroX, gyroY, gyroZ ); } else { gyroX = 0; gyroY = 0; gyroZ = 0; } // =================================================== // REMAPOWANIE GAZELA // =================================================== // ACC AX = -sensorY; AY = -sensorX; AZ = sensorZ; // GYRO GX = -gyroY; GY = -gyroX; GZ = gyroZ; // =================================================== // G // =================================================== G = sqrt( AX * AX + AY * AY + AZ * AZ ); // =================================================== // ANGLE // =================================================== Angle = atan2( sqrt( AX * AX + AY * AY ), AZ ) * 180.0 / PI; // =================================================== // ANGLEY // =================================================== unsigned long currentTime = micros(); if (previousTime != 0) { float dt = (currentTime - previousTime) / 1000000.0; angleY_Gyro += GY * dt; } previousTime = currentTime; AngleY = angleY_Gyro; } // ===================================================== // WYSŁANIE LIVE // ===================================================== void sendLiveData() { float AX; float AY; float AZ; float G; float Angle; float GX; float GY; float GZ; float AngleY; readSensorData( AX, AY, AZ, G, Angle, GX, GY, GZ, AngleY ); char buffer[100]; snprintf( buffer, sizeof(buffer), "LIVE,%.4f,%.4f,%.4f,%.4f,%.2f,%.4f,%.4f,%.4f,%.2f", AX, AY, AZ, G, Angle, GX, GY, GZ, AngleY ); // --------------------------------------------------- // BLE // --------------------------------------------------- liveCharacteristic.writeValue( buffer ); // --------------------------------------------------- // SERIAL MONITOR // --------------------------------------------------- Serial.println( buffer ); } // ===================================================== // OBSŁUGA COMMAND // ===================================================== void processCommand() { if ( commandCharacteristic.written() ) { String command = commandCharacteristic.value(); command.trim(); Serial.print( "COMMAND → " ); Serial.println( command ); // ================================================= // L // ================================================= if ( command == "L" || command == "l" ) { liveMode = true; angleY_Gyro = 0.0; previousTime = micros(); lastSampleTime = millis(); Serial.println( "COMMAND L RECEIVED" ); Serial.println( "LIVE START" ); } // ================================================= // Q // ================================================= if ( command == "Q" || command == "q" ) { liveMode = false; Serial.println( "COMMAND Q RECEIVED" ); Serial.println( "LIVE STOP" ); } } } // ===================================================== // SETUP // ===================================================== void setup() { Serial.begin(115200); while (!Serial) { delay(10); } // --------------------------------------------------- // SENSOR // --------------------------------------------------- if (!IMU.begin()) { Serial.println( "ERROR,IMU_NOT_FOUND" ); while (1) { delay(1000); } } // --------------------------------------------------- // BLE // --------------------------------------------------- if (!BLE.begin()) { Serial.println( "ERROR,BLE_NOT_FOUND" ); while (1) { delay(1000); } } // --------------------------------------------------- // DEVICE NAME // --------------------------------------------------- BLE.setLocalName( "GAZELA SENSOR" ); BLE.setDeviceName( "GAZELA SENSOR" ); // --------------------------------------------------- // ADVERTISED SERVICE // --------------------------------------------------- BLE.setAdvertisedService( sensorService ); // --------------------------------------------------- // SENSOR SERVICE // --------------------------------------------------- sensorService.addCharacteristic( liveCharacteristic ); // --------------------------------------------------- // CONTROL SERVICE // --------------------------------------------------- controlService.addCharacteristic( commandCharacteristic ); // --------------------------------------------------- // ADD SERVICES // --------------------------------------------------- BLE.addService( sensorService ); BLE.addService( controlService ); // --------------------------------------------------- // INITIAL LIVE VALUE // --------------------------------------------------- liveCharacteristic.writeValue( "LIVE,0,0,0,0,0,0,0,0,0" ); // --------------------------------------------------- // START BLE // --------------------------------------------------- BLE.advertise(); // --------------------------------------------------- // INITIAL STATE // --------------------------------------------------- liveMode = false; angleY_Gyro = 0.0; previousTime = micros(); // --------------------------------------------------- // SERIAL // --------------------------------------------------- Serial.println( "GAZELA SENSOR BLE TEST v5" ); Serial.println( "BLE READY" ); Serial.println( "ADVERTISING" ); Serial.println( "LIVE OFF" ); } // ===================================================== // LOOP // ===================================================== void loop() { // --------------------------------------------------- // BLE POLL // --------------------------------------------------- BLE.poll(); // --------------------------------------------------- // COMMAND // --------------------------------------------------- processCommand(); // --------------------------------------------------- // CENTRAL // --------------------------------------------------- BLEDevice central = BLE.central(); if (central) { Serial.print( "CONNECTED: " ); Serial.println( central.address() ); // ------------------------------------------------- // CONNECTION LOOP // ------------------------------------------------- while ( central.connected() ) { unsigned long now = millis(); // ----------------------------------------------- // COMMAND // ----------------------------------------------- processCommand(); // ----------------------------------------------- // LIVE // ----------------------------------------------- if ( liveMode ) { if ( now - lastSampleTime >= SAMPLE_INTERVAL ) { lastSampleTime = now; sendLiveData(); } } BLE.poll(); } // ------------------------------------------------- // DISCONNECTED // ------------------------------------------------- Serial.println( "DISCONNECTED" ); liveMode = false; angleY_Gyro = 0.0; previousTime = micros(); } }
